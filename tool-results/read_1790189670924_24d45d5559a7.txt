   150→// ============================================================
   151→// التطبيق الرئيسي
   152→// ============================================================
   153→export default function AdminApp({ onExit }: { onExit: () => void }) {
   154→  const { toast } = useToast()
   155→
   156→  // درع الـ PIN والجلسة
   157→  const [pin, setPin] = useState('')
   158→  const [unlocked, setUnlocked] = useState(false)
   159→  const [checking, setChecking] = useState(false)
   160→  const [errorNonce, setErrorNonce] = useState(0)
   161→
   162→  // بيانات الغرفة
   163→  const [summary, setSummary] = useState<AccountingSummary | null>(null)
   164→  const [vouchersData, setVouchersData] = useState<VouchersPayload | null>(null)
   165→  const [orders, setOrders] = useState<OrderDTO[]>([])
   166→  const [promos, setPromos] = useState<PromoAdmin[] | null>(null)
   167→  const [giftCodes, setGiftCodes] = useState<TopupCodeAdmin[] | null>(null)
   168→  const [feed, setFeed] = useState<FeedEvent[]>([])
   169→  const [positions, setPositions] = useState<Record<string, FleetPos>>({})
   170→  const [live, setLive] = useState(false)
   171→  const [lastSync, setLastSync] = useState<number | null>(null)
   172→  const [busy, setBusy] = useState<string | null>(null)
   173→  const [tab, setTab] = useState('dashboard')
   174→
   175→  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null)
   176→
   177→  // ---------- سجل الأحداث الحي ----------
   178→  const pushEvent = useCallback((kind: FeedEvent['kind'], text: string, sub?: string) => {
   179→    setFeed((prev) =>
   180→      [
   181→        {
   182→          id: `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   183→          at: Date.now(),
   184→          kind,
   185→          text,
   186→          sub,
   187→        },
   188→        ...prev,
   189→      ].slice(0, 12),
   190→    )
   191→  }, [])
   192→
   193→  // ---------- قفل الجلسة ----------
   194→  const lockSession = useCallback(
   195→    (message?: string) => {
   196→      setUnlocked(false)
   197→      setPin('')
   198→      setSummary(null)
   199→      setVouchersData(null)
   200→      setOrders([])
   201→      setPromos(null)
   202→      setGiftCodes(null)
   203→      setPositions({})
   204→      setFeed([])
   205→      setLive(false)
   206→      setLastSync(null)
   207→      setBusy(null)
   208→      if (message) {
   209→        toast({ title: '🔒 أُقفلت الجلسة الإدارية', description: message, variant: 'destructive' })
   210→      }
   211→    },
   212→    [toast],
   213→  )
   214→
   215→  // ---------- معالج أخطاء موحد (401 = قفل فوري) ----------
   216→  const handleAdminError = useCallback(
   217→    (e: unknown) => {
   218→      if (e instanceof ApiError && e.status === 401) {
   219→        lockSession('رمز الإدارة غير صحيح أو منتهٍ — يلزم إعادة فتح الغرفة')
   220→        return
   221→      }
   222→      toast({
   223→        title: 'تعذر تنفيذ العملية',
   224→        description: e instanceof Error ? e.message : String(e),
   225→        variant: 'destructive',
   226→      })
   227→    },
   228→    [lockSession, toast],
   229→  )
   230→
   231→  // ---------- تحميل بيانات الغرفة (ملخص + سندات + طلبات) ----------
   232→  const loadAll = useCallback(
   233→    async (currentPin: string) => {
   234→      if (!currentPin) return
   235→      const headers = { 'x-admin-pin': currentPin }
   236→      const [sumRes, vouRes, ordRes, promoRes, giftRes] = await Promise.allSettled([
   237→        api<AccountingSummary>('/api/accounting/summary', { headers }),
   238→        api<VouchersPayload>('/api/vouchers', { headers }),
   239→        apiGet<{ orders: OrderDTO[] }>('/api/orders?status=all&limit=50'),
   240→        api<{ promos: PromoAdmin[] }>('/api/admin/promos', { headers }),
   241→        api<{ codes: TopupCodeAdmin[] }>('/api/admin/topup-codes', { headers }),
   242→      ])
   243→
   244→      for (const r of [sumRes, vouRes]) {
   245→        if (r.status === 'rejected' && r.reason instanceof ApiError && r.reason.status === 401) {
   246→          lockSession('انتهت صلاحية جلسة الإدارة — رمز غير صحيح')
   247→          return
   248→        }
   249→      }
   250→
   251→      if (sumRes.status === 'fulfilled') setSummary(sumRes.value)
   252→      if (vouRes.status === 'fulfilled') setVouchersData(vouRes.value)
   253→      if (ordRes.status === 'fulfilled') setOrders(ordRes.value.orders ?? [])
   254→      if (promoRes.status === 'fulfilled') setPromos(promoRes.value.promos ?? [])
   255→      if (giftRes.status === 'fulfilled') setGiftCodes(giftRes.value.codes ?? [])
   256→      setLastSync(Date.now())
   257→    },
   258→    [lockSession],
   259→  )
   260→
   261→  // ---------- فتح الجلسة بالـ PIN ----------
   262→  const attemptUnlock = useCallback(
   263→    async (code: string) => {
   264→      if (code.length !== 4 || checking) return
   265→      setChecking(true)
   266→      try {
   267→        const s = await api<AccountingSummary>('/api/accounting/summary', {
   268→          headers: { 'x-admin-pin': code },
   269→        })
   270→        setPin(code)
   271→        setSummary(s)
   272→        setErrorNonce(0)
   273→        setUnlocked(true)
   274→        setFeed([
   275→          {
   276→            id: `unlock-${Date.now()}`,
   277→            at: Date.now(),
   278→            kind: 'status',
   279→            text: '🔓 فُتحت جلسة غرفة العمليات — البث المالي واللوجستي يعمل الآن',
   280→          },
   281→        ])
   282→        toast({ title: 'أهلاً بك في غرفة العمليات 🛡️', description: 'كل قرش مُساءل — البث الحي متصل' })
   283→      } catch (e) {
   284→        setPin('')
   285→        if (e instanceof ApiError && e.status === 401) {
   286→          setErrorNonce((n) => n + 1)
   287→          toast({ title: 'رمز غير صحيح', description: 'رمز PIN الإداري مرفوض — حاول مجدداً', variant: 'destructive' })
   288→        } else {
   289→          toast({
   290→            title: 'تعذر الاتصال بالخادم',
   291→            description: e instanceof Error ? e.message : 'تحقق من الشبكة وحاول مجدداً',
   292→            variant: 'destructive',
   293→          })
   294→        }
   295→      } finally {
   296→        setChecking(false)
   297→      }
   298→    },
   299→    [checking, toast],
   300→  )
   301→
   302→  // ---------- دورة البيانات: فورية + كل 8 ثوانٍ ----------
   303→  useEffect(() => {
   304→    if (!unlocked || !pin) return
   305→    void loadAll(pin)
   306→    pollRef.current = setInterval(() => {
   307→      void loadAll(pin)
   308→    }, POLL_MS)
   309→    return () => {
   310→      if (pollRef.current) {
   311→        clearInterval(pollRef.current)
   312→        pollRef.current = null
   313→      }
   314→    }
   315→  }, [unlocked, pin, loadAll])
   316→
   317→  // ---------- البث اللحظي: غرفة admin:fleet ----------
   318→  useEffect(() => {
   319→    if (!unlocked) return
   320→    const s = getSocket()
   321→
   322→    const refreshMoney = () => {
   323→      void loadAll(pin)
   324→    }
   325→
   326→    const onConnect = () => {
   327→      setLive(true)
   328→      joinRooms(['admin:fleet'])
   329→      s.emit('fleet:snapshot_request')
   330→    }
   331→    const onDisconnect = () => setLive(false)
   332→
   333→    const onSnapshot = (list: FleetPos[]) => {
   334→      if (!Array.isArray(list)) return
   335→      setPositions((prev) => {
   336→        const next = { ...prev }
   337→        for (const rec of list) if (rec?.driverId) next[rec.driverId] = rec
   338→        return next
   339→      })
   340→    }
   341→
   342→    const onDriverLocation = (rec: FleetPos) => {
   343→      if (!rec?.driverId) return
   344→      setPositions((prev) => ({ ...prev, [rec.driverId]: rec }))
   345→    }
   346→
   347→    const onDriverStatus = (d: { driverId: string; fullName: string; status: DriverDTO['status'] }) => {
   348→      if (!d?.driverId) return
   349→      setPositions((prev) => {
   350→        if (!prev[d.driverId]) return prev
   351→        return {
   352→          ...prev,
   353→          [d.driverId]: { ...prev[d.driverId], status: d.status, updatedAt: new Date().toISOString() },
   354→        }
   355→      })
   356→      setSummary((prev) =>
   357→        prev
   358→          ? { ...prev, drivers: prev.drivers.map((x) => (x.id === d.driverId ? { ...x, status: d.status } : x)) }
   359→          : prev,
   360→      )
   361→      pushEvent('status', `🛵 الكابتن ${d.fullName} أصبح «${STATUS_LABELS[d.status] ?? d.status}»`)
   362→    }
   363→
   364→    const onOrderStatus = (d: { orderId: string; status: string; orderNumber: string }) => {
   365→      if (!d?.orderNumber) return
   366→      pushEvent('status', `📦 الطلب ${d.orderNumber}: ${STATUS_LABELS[d.status] ?? d.status}`)
   367→      setOrders((prev) =>
   368→        prev.map((o) => (o.id === d.orderId ? { ...o, status: d.status as OrderDTO['status'] } : o)),
   369→      )
   370→    }
   371→
   372→    const onRadar = (d: { orderId: string; orderNumber: string; storeName: string; candidates: number }) => {
   373→      if (!d?.orderNumber) return
   374→      pushEvent(
   375→        'radar',
   376→        `📡 رادار: أُرسل عرض الطلب ${d.orderNumber} من «${d.storeName}» إلى ${d.candidates ?? 0} كابتن`,
   377→      )
   378→    }
   379→
   380→    const onCustodySettled = (d: { driverName: string; amount: number; voucherNumber: string }) => {
   381→      if (!d?.voucherNumber) return
   382→      pushEvent(
   383→        'custody',
   384→        `💰 توريد عهدة الكابتن ${d.driverName} بمبلغ ${(d.amount ?? 0).toFixed(2)} د.ل`,
   385→        `سند قبض ${d.voucherNumber}`,
   386→      )
   387→      refreshMoney()
   388→    }
   389→
   390→    const onStorePayout = (d: { storeName: string; amount: number; voucherNumber: string }) => {
   391→      if (!d?.voucherNumber) return
   392→      pushEvent(
   393→        'payout',
   394→        `🏪 صرف مستحقات متجر «${d.storeName}» بمبلغ ${(d.amount ?? 0).toFixed(2)} د.ل`,
   395→        `سند صرف ${d.voucherNumber}`,
   396→      )
   397→      refreshMoney()
   398→    }
   399→
   400→    const onPromoApplied = (d: { orderNumber?: string; promoCode?: string; promoLabel?: string; promoKind?: string; discountLyd?: number; cashbackLyd?: number; customerName?: string }) => {
   401→      if (!d?.promoCode) return
   402→      if (d.promoKind === 'cashback') {
   403→        pushEvent(
   404→          'promo',
   405→          `💸 كاش باك ${d.promoLabel ?? d.promoCode} على طلب ${d.orderNumber ?? ''} للزبون ${d.customerName ?? ''}`,
   406→          `سيُودع ${(d.cashbackLyd ?? 0).toFixed(2)} د.ل محفظته عند التسليم — من ميزانية تسويق المنصة`,
   407→        )
   408→      } else {
   409→        pushEvent(
   410→          'promo',
   411→          `🎟️ استخدم الزبون ${d.customerName ?? ''} كوبون ${d.promoLabel ?? d.promoCode} على الطلب ${d.orderNumber ?? ''}`,
   412→          `وفّرنا له ${(d.discountLyd ?? 0).toFixed(2)} د.ل — على حساب المنصة`,
   413→        )
   414→      }
   415→    }
   416→
   417→    const onTopupRedeemed = (d: { phone?: string; code?: string; amountLyd?: number; balance?: number }) => {
   418→      if (!d?.code) return
   419→      pushEvent(
   420→        'topup',
   421→        `🎁 شُحنت محفظة ${d.phone ?? ''} بكود ${d.code}`,
   422→        `قيمة ${(d.amountLyd ?? 0).toFixed(2)} د.ل — الرصيد الآن ${(d.balance ?? 0).toFixed(2)} د.ل`,
   423→      )
   424→    }
   425→
   426→    const onGiftCardSold = (d: { orderNumber?: string; codesCount?: number; totalValueLyd?: number }) => {
   427→      if (!d?.orderNumber) return
   428→      pushEvent(
   429→        'topup',
   430→        `💳 صُدرت ${d.codesCount ?? 0} بطاقة شحن من واصل متاع — طلب ${d.orderNumber}`,
   431→        `قيمة إجمالية ${(d.totalValueLyd ?? 0).toFixed(2)} د.ل — التزام على المنصة حتى استخدام الأكواد`,
   432→      )
   433→    }
   434→
   435→    const onPromoCashback = (d: { orderNumber?: string; customerPhone?: string; amountLyd?: number }) => {
   436→      if (!d?.orderNumber) return
   437→      pushEvent(
   438→        'promo',
   439→        `💸 كاش باك ${(d.amountLyd ?? 0).toFixed(2)} د.ل وُدع محفظة ${d.customerPhone ?? 'الزبون'} من طلب ${d.orderNumber}`,
   440→        'كوبون كاش باك — على حساب ميزانية تسويق المنصة',
   441→      )
   442→    }
   443→
   444→    if (s.connected) onConnect()
   445→    s.on('connect', onConnect)
   446→    s.on('disconnect', onDisconnect)
   447→    s.on('fleet:snapshot', onSnapshot)
   448→    s.on('fleet:driver_location', onDriverLocation)
   449→    s.on('fleet:driver_status', onDriverStatus)
   450→    s.on('order:status_changed', onOrderStatus)
   451→    s.on('order:radar_broadcast', onRadar)
   452→    s.on('driver:custody_settled', onCustodySettled)
   453→    s.on('store:payout_settled', onStorePayout)
   454→    s.on('store:promo_applied', onPromoApplied)
   455→    s.on('wallet:topup_redeemed', onTopupRedeemed)
   456→    s.on('store:giftcard_sold', onGiftCardSold)
   457→    s.on('wallet:promo_cashback', onPromoCashback)
   458→
   459→    return () => {
   460→      s.off('connect', onConnect)
   461→      s.off('disconnect', onDisconnect)
   462→      s.off('fleet:snapshot', onSnapshot)
   463→      s.off('fleet:driver_location', onDriverLocation)
   464→      s.off('fleet:driver_status', onDriverStatus)
   465→      s.off('order:status_changed', onOrderStatus)
   466→      s.off('order:radar_broadcast', onRadar)
   467→      s.off('driver:custody_settled', onCustodySettled)
   468→      s.off('store:payout_settled', onStorePayout)
   469→      s.off('store:promo_applied', onPromoApplied)
   470→      s.off('wallet:topup_redeemed', onTopupRedeemed)
   471→      s.off('store:giftcard_sold', onGiftCardSold)
   472→      s.off('wallet:promo_cashback', onPromoCashback)
   473→      leaveRooms(['admin:fleet'])
   474→    }
   475→  }, [unlocked, pin, loadAll, pushEvent])
