   156→const REJECT_REASON_LABELS: Record<string, string> = {
   157→  exhausted: 'نفدت الحصة',

   162→export default function AdminApp({ onExit }: { onExit: () => void }) {
   163→  const { toast } = useToast()
   164→
   165→  // درع الـ PIN والجلسة
   166→  const [pin, setPin] = useState('')
   167→  const [unlocked, setUnlocked] = useState(false)
   168→  const [checking, setChecking] = useState(false)
   169→  const [errorNonce, setErrorNonce] = useState(0)
   170→
   171→  // بيانات الغرفة
   172→  const [summary, setSummary] = useState<AccountingSummary | null>(null)
   173→  const [vouchersData, setVouchersData] = useState<VouchersPayload | null>(null)
   174→  const [orders, setOrders] = useState<OrderDTO[]>([])
   175→  const [promos, setPromos] = useState<PromoAdmin[] | null>(null)
   176→  const [giftCodes, setGiftCodes] = useState<TopupCodeAdmin[] | null>(null)
   177→  const [feed, setFeed] = useState<FeedEvent[]>([])
   178→  const [positions, setPositions] = useState<Record<string, FleetPos>>({})
   179→  const [live, setLive] = useState(false)
   180→  const [lastSync, setLastSync] = useState<number | null>(null)
   181→  const [busy, setBusy] = useState<string | null>(null)
   182→  const [tab, setTab] = useState('dashboard')
   183→
   184→  // 🚨 تنبيه الفرص الضائعة اللحظي (كوبون رفض زبون أثناء الدفع)
   185→  const [promoAlert, setPromoAlert] = useState<{
   186→    code: string | null
   187→    label: string | null
   188→    reason: string
   189→    lostLyd: number
   190→    subtotalLyd: number
   191→    phoneMasked: string
   192→    at: number
   193→  } | null>(null)
   194→  const promoAlertTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   195→
   196→  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null)
   197→
   198→  // ---------- سجل الأحداث الحي ----------
   199→  const pushEvent = useCallback((kind: FeedEvent['kind'], text: string, sub?: string) => {
   200→    setFeed((prev) =>
   201→      [
   202→        {
   203→          id: `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   204→          at: Date.now(),
   205→          kind,
   206→          text,
   207→          sub,
   208→        },
   209→        ...prev,
   210→      ].slice(0, 12),
   211→    )
   212→  }, [])
   213→
   214→  // ---------- قفل الجلسة ----------
   215→  const lockSession = useCallback(
   216→    (message?: string) => {
   217→      setUnlocked(false)
   218→      setPin('')
   219→      setSummary(null)
   220→      setVouchersData(null)
   221→      setOrders([])
   222→      setPromos(null)
   223→      setGiftCodes(null)
   224→      setPositions({})
   225→      setFeed([])
   226→      setLive(false)
   227→      setLastSync(null)
   228→      setBusy(null)
   229→      if (message) {
   230→        toast({ title: '🔒 أُقفلت الجلسة الإدارية', description: message, variant: 'destructive' })
   231→      }
   232→    },
   233→    [toast],
   234→  )
   235→
   236→  // ---------- معالج أخطاء موحد (401 = قفل فوري) ----------
   237→  const handleAdminError = useCallback(
   238→    (e: unknown) => {
   239→      if (e instanceof ApiError && e.status === 401) {
   240→        lockSession('رمز الإدارة غير صحيح أو منتهٍ — يلزم إعادة فتح الغرفة')
   241→        return
   242→      }
   243→      toast({
   244→        title: 'تعذر تنفيذ العملية',
   245→        description: e instanceof Error ? e.message : String(e),
   246→        variant: 'destructive',
   247→      })
   248→    },
   249→    [lockSession, toast],
   250→  )
   251→
   252→  // ---------- تحميل بيانات الغرفة (ملخص + سندات + طلبات) ----------
   253→  const loadAll = useCallback(
   254→    async (currentPin: string) => {
   255→      if (!currentPin) return
   256→      const headers = { 'x-admin-pin': currentPin }
   257→      const [sumRes, vouRes, ordRes, promoRes, giftRes] = await Promise.allSettled([
   258→        api<AccountingSummary>('/api/accounting/summary', { headers }),
   259→        api<VouchersPayload>('/api/vouchers', { headers }),
   260→        apiGet<{ orders: OrderDTO[] }>('/api/orders?status=all&limit=50'),
   261→        api<{ promos: PromoAdmin[] }>('/api/admin/promos', { headers }),
   262→        api<{ codes: TopupCodeAdmin[] }>('/api/admin/topup-codes', { headers }),
   263→      ])
   264→
   265→      for (const r of [sumRes, vouRes]) {
   266→        if (r.status === 'rejected' && r.reason instanceof ApiError && r.reason.status === 401) {
   267→          lockSession('انتهت صلاحية جلسة الإدارة — رمز غير صحيح')
   268→          return
   269→        }
   270→      }
   271→
   272→      if (sumRes.status === 'fulfilled') setSummary(sumRes.value)
   273→      if (vouRes.status === 'fulfilled') setVouchersData(vouRes.value)
   274→      if (ordRes.status === 'fulfilled') setOrders(ordRes.value.orders ?? [])
   275→      if (promoRes.status === 'fulfilled') setPromos(promoRes.value.promos ?? [])
   276→      if (giftRes.status === 'fulfilled') setGiftCodes(giftRes.value.codes ?? [])
   277→      setLastSync(Date.now())
   278→    },
   279→    [lockSession],
   280→  )
   281→
   282→  // ---------- فتح الجلسة بالـ PIN ----------
   283→  const attemptUnlock = useCallback(
   284→    async (code: string) => {
   285→      if (code.length !== 4 || checking) return
   286→      setChecking(true)
   287→      try {
   288→        const s = await api<AccountingSummary>('/api/accounting/summary', {
   289→          headers: { 'x-admin-pin': code },
   290→        })
   291→        setPin(code)
   292→        setSummary(s)
   293→        setErrorNonce(0)
   294→        setUnlocked(true)
   295→        setFeed([
   296→          {
   297→            id: `unlock-${Date.now()}`,
   298→            at: Date.now(),
   299→            kind: 'status',
   300→            text: '🔓 فُتحت جلسة غرفة العمليات — البث المالي واللوجستي يعمل الآن',
   301→          },
   302→        ])
   303→        toast({ title: 'أهلاً بك في غرفة العمليات 🛡️', description: 'كل قرش مُساءل — البث الحي متصل' })
   304→      } catch (e) {
   305→        setPin('')
   306→        if (e instanceof ApiError && e.status === 401) {
   307→          setErrorNonce((n) => n + 1)
   308→          toast({ title: 'رمز غير صحيح', description: 'رمز PIN الإداري مرفوض — حاول مجدداً', variant: 'destructive' })
   309→        } else {
   310→          toast({
   311→            title: 'تعذر الاتصال بالخادم',
   312→            description: e instanceof Error ? e.message : 'تحقق من الشبكة وحاول مجدداً',
   313→            variant: 'destructive',
   314→          })
   315→        }
   316→      } finally {
   317→        setChecking(false)
   318→      }
   319→    },
   320→    [checking, toast],
   321→  )
   322→
   323→  // ---------- دورة البيانات: فورية + كل 8 ثوانٍ ----------
   324→  useEffect(() => {
   325→    if (!unlocked || !pin) return
   326→    void loadAll(pin)
   327→    pollRef.current = setInterval(() => {
   328→      void loadAll(pin)
   329→    }, POLL_MS)
   330→    return () => {
   331→      if (pollRef.current) {
   332→        clearInterval(pollRef.current)
   333→        pollRef.current = null
   334→      }
   335→    }
   336→  }, [unlocked, pin, loadAll])
   337→
   338→  // ---------- البث اللحظي: غرفة admin:fleet ----------
   339→  useEffect(() => {
   340→    if (!unlocked) return
   341→    const s = getSocket()
   342→
   343→    const refreshMoney = () => {
   344→      void loadAll(pin)
   345→    }
   346→
   347→    const onConnect = () => {
   348→      setLive(true)
   349→      joinRooms(['admin:fleet'])
   350→      s.emit('fleet:snapshot_request')
   351→    }
   352→    const onDisconnect = () => setLive(false)
   353→
   354→    const onSnapshot = (list: FleetPos[]) => {
   355→      if (!Array.isArray(list)) return
   356→      setPositions((prev) => {
   357→        const next = { ...prev }
   358→        for (const rec of list) if (rec?.driverId) next[rec.driverId] = rec
   359→        return next
   360→      })
   361→    }
   362→
   363→    const onDriverLocation = (rec: FleetPos) => {
   364→      if (!rec?.driverId) return
   365→      setPositions((prev) => ({ ...prev, [rec.driverId]: rec }))
   366→    }
   367→
   368→    const onDriverStatus = (d: { driverId: string; fullName: string; status: DriverDTO['status'] }) => {
   369→      if (!d?.driverId) return
   370→      setPositions((prev) => {
   371→        if (!prev[d.driverId]) return prev
   372→        return {
   373→          ...prev,
   374→          [d.driverId]: { ...prev[d.driverId], status: d.status, updatedAt: new Date().toISOString() },
   375→        }
   376→      })
   377→      setSummary((prev) =>
   378→        prev
   379→          ? { ...prev, drivers: prev.drivers.map((x) => (x.id === d.driverId ? { ...x, status: d.status } : x)) }
   380→          : prev,
   381→      )
   382→      pushEvent('status', `🛵 الكابتن ${d.fullName} أصبح «${STATUS_LABELS[d.status] ?? d.status}»`)
   383→    }
   384→
   385→    const onOrderStatus = (d: { orderId: string; status: string; orderNumber: string }) => {
   386→      if (!d?.orderNumber) return
   387→      pushEvent('status', `📦 الطلب ${d.orderNumber}: ${STATUS_LABELS[d.status] ?? d.status}`)
   388→      setOrders((prev) =>
   389→        prev.map((o) => (o.id === d.orderId ? { ...o, status: d.status as OrderDTO['status'] } : o)),
   390→      )
   391→    }
   392→
   393→    const onRadar = (d: { orderId: string; orderNumber: string; storeName: string; candidates: number }) => {
   394→      if (!d?.orderNumber) return
   395→      pushEvent(
   396→        'radar',
   397→        `📡 رادار: أُرسل عرض الطلب ${d.orderNumber} من «${d.storeName}» إلى ${d.candidates ?? 0} كابتن`,
   398→      )
   399→    }
   400→
   401→    const onCustodySettled = (d: { driverName: string; amount: number; voucherNumber: string }) => {
   402→      if (!d?.voucherNumber) return
   403→      pushEvent(
   404→        'custody',
   405→        `💰 توريد عهدة الكابتن ${d.driverName} بمبلغ ${(d.amount ?? 0).toFixed(2)} د.ل`,
   406→        `سند قبض ${d.voucherNumber}`,
   407→      )
   408→      refreshMoney()
   409→    }
   410→
   411→    const onStorePayout = (d: { storeName: string; amount: number; voucherNumber: string }) => {
   412→      if (!d?.voucherNumber) return
   413→      pushEvent(
   414→        'payout',
   415→        `🏪 صرف مستحقات متجر «${d.storeName}» بمبلغ ${(d.amount ?? 0).toFixed(2)} د.ل`,
   416→        `سند صرف ${d.voucherNumber}`,
   417→      )
   418→      refreshMoney()
   419→    }
   420→
   421→    const onPromoApplied = (d: { orderNumber?: string; promoCode?: string; promoLabel?: string; promoKind?: string; discountLyd?: number; cashbackLyd?: number; customerName?: string }) => {
   422→      if (!d?.promoCode) return
   423→      if (d.promoKind === 'cashback') {
   424→        pushEvent(
   425→          'promo',
   426→          `💸 كاش باك ${d.promoLabel ?? d.promoCode} على طلب ${d.orderNumber ?? ''} للزبون ${d.customerName ?? ''}`,
   427→          `سيُودع ${(d.cashbackLyd ?? 0).toFixed(2)} د.ل محفظته عند التسليم — من ميزانية تسويق المنصة`,
   428→        )
   429→      } else {
   430→        pushEvent(
   431→          'promo',
   432→          `🎟️ استخدم الزبون ${d.customerName ?? ''} كوبون ${d.promoLabel ?? d.promoCode} على الطلب ${d.orderNumber ?? ''}`,
   433→          `وفّرنا له ${(d.discountLyd ?? 0).toFixed(2)} د.ل — على حساب المنصة`,
   434→        )
   435→      }
   436→    }
   437→
   438→    const onTopupRedeemed = (d: { phone?: string; code?: string; amountLyd?: number; balance?: number }) => {
   439→      if (!d?.code) return
   440→      pushEvent(
   441→        'topup',
   442→        `🎁 شُحنت محفظة ${d.phone ?? ''} بكود ${d.code}`,
   443→        `قيمة ${(d.amountLyd ?? 0).toFixed(2)} د.ل — الرصيد الآن ${(d.balance ?? 0).toFixed(2)} د.ل`,
   444→      )
   445→    }
   446→
   447→    const onGiftCardSold = (d: { orderNumber?: string; codesCount?: number; totalValueLyd?: number }) => {
   448→      if (!d?.orderNumber) return
   449→      pushEvent(
   450→        'topup',
   451→        `💳 صُدرت ${d.codesCount ?? 0} بطاقة شحن من واصل متاع — طلب ${d.orderNumber}`,
   452→        `قيمة إجمالية ${(d.totalValueLyd ?? 0).toFixed(2)} د.ل — التزام على المنصة حتى استخدام الأكواد`,
   453→      )
   454→    }
   455→
   456→    const onPromoCashback = (d: { orderNumber?: string; customerPhone?: string; amountLyd?: number }) => {
   457→      if (!d?.orderNumber) return
   458→      pushEvent(
   459→        'promo',
   460→        `💸 كاش باك ${(d.amountLyd ?? 0).toFixed(2)} د.ل وُدع محفظة ${d.customerPhone ?? 'الزبون'} من طلب ${d.orderNumber}`,
   461→        'كوبون كاش باك — على حساب ميزانية تسويق المنصة',
   462→      )
   463→    }
   464→
   465→    const onPromoReject = (d: {
   466→      code?: string | null
   467→      label?: string | null
   468→      reason?: string
   469→      lostLyd?: number
   470→      subtotalLyd?: number
   471→      phoneMasked?: string
   472→    }) => {
   473→      const reason = d?.reason ?? 'unknown'
   474→      if (reason === 'unknown') return
   475→      setPromoAlert({
   476→        code: d?.code ?? null,
   477→        label: d?.label ?? null,
   478→        reason,
   479→        lostLyd: d?.lostLyd ?? 0,
   480→        subtotalLyd: d?.subtotalLyd ?? 0,
   481→        phoneMasked: d?.phoneMasked ?? '',
   482→        at: Date.now(),
   483→      })
