    50→function promoValueLabel(p: PromoAdmin): string {
    51→  if (p.kind === 'cashback') {
    52→    const cap = p.maxDiscountLyd != null ? ` حتى سقف ${p.maxDiscountLyd.toFixed(2)} د.ل` : ''
    53→    return p.percent != null && p.percent > 0 ? `إيداع ${p.percent}%${cap}` : `إيداع ${(p.flatLyd ?? 0).toFixed(2)} د.ل`
    54→  }
    55→  if (p.kind === 'flat') return `خصم ثابت ${(p.flatLyd ?? 0).toFixed(2)} د.ل`
    56→  const cap = p.maxDiscountLyd != null ? ` حتى سقف ${p.maxDiscountLyd.toFixed(2)} د.ل` : ''
    57→  return `خصم ${p.percent}%${cap}`
    58→}

   172→function TopupCodeRow({
   173→  code,
   174→  busy,
   175→  onToggle,
   176→}: {
   177→  code: TopupCodeAdmin
   178→  busy: boolean
   179→  onToggle: (next: boolean) => void
   180→}) {
   181→  const { toast } = useToast()
   182→  const used = !!code.usedByPhone
   183→  const stateLabel = used ? 'مستهلك' : code.isActive ? 'نشط' : 'مُعطّل'
   184→  const stateClass = used
   185→    ? 'bg-[rgba(147,142,126,0.15)] text-[#938E7E]'
   186→    : code.isActive
   187→      ? 'bg-[rgba(16,185,129,0.15)] text-[#10B981]'
   188→      : 'bg-[rgba(245,158,11,0.15)] text-[#F59E0B]'
   189→
   190→  const copy = async () => {
   191→    try {
   192→      await navigator.clipboard.writeText(code.code)
   193→      toast({ title: '📋 نُسخ الكود', description: code.code })
   194→    } catch {
   195→      toast({ title: code.code, description: 'انسخ الكود يدوياً' })
   196→    }
   197→  }
   198→
   199→  return (
   200→    <div className="flex flex-wrap items-center gap-3 rounded-xl border border-[rgba(212,175,55,0.18)] bg-[#041710] px-3 py-2.5">
   201→      <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-[#0A3324] text-lg">
   202→        🎁
   203→      </span>
   204→      <div className="min-w-0 flex-1">
   205→        <p className="font-money ltr-nums text-xs font-black tracking-wider text-[#F5D061]" dir="ltr">
   206→          {code.code}
   207→        </p>
   208→        <p className="mt-0.5 text-[10px] text-[#938E7E]">
   209→          {used ? (
   210→            <>
   211→              استهلكه <span dir="ltr" className="font-money">{code.usedByPhone}</span>
   212→            </>
   213→          ) : (
   214→            code.note || 'كود شحن محفظة — يُستخدم مرة واحدة'
   215→          )}
   216→          {' • '}
   217→          <span className="font-money ltr-nums">{fmtDateTime(code.createdAt)}</span>
   218→        </p>
   219→      </div>
   220→      <Ly amount={code.amountLyd} bold className="text-sm text-[#10B981]" />
   221→      <Badge className={`border-0 px-2 py-0.5 text-[9px] font-bold ${stateClass}`}>{stateLabel}</Badge>
   222→      <div className="flex items-center gap-1.5">
   223→        <button
   224→          onClick={copy}
   225→          aria-label={`نسخ الكود ${code.code}`}
   226→          className="flex h-11 w-11 items-center justify-center rounded-lg text-[#938E7E] transition hover:bg-[rgba(212,175,55,0.12)] hover:text-[#F5D061]"
   227→        >
   228→          <Copy className="h-4 w-4" />
   229→        </button>
   230→        {!used && (
   231→          <Switch
   232→            checked={code.isActive}
   233→            disabled={busy}
   234→            onCheckedChange={onToggle}
   235→            aria-label={`${code.isActive ? 'تعطيل' : 'تفعيل'} كود الشحن ${code.code}`}
   236→          />
   237→        )}
   238→      </div>
   239→    </div>
   240→  )
   241→}
