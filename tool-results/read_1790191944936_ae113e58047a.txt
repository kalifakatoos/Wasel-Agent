    66→function promoValueLabel(p: PromoAdmin): string {
    67→  if (p.kind === 'cashback') {
    68→    const cap = p.maxDiscountLyd != null ? ` حتى سقف ${p.maxDiscountLyd.toFixed(2)} د.ل` : ''
    69→    return p.percent != null && p.percent > 0 ? `إيداع ${p.percent}%${cap}` : `إيداع ${(p.flatLyd ?? 0).toFixed(2)} د.ل`
    70→  }
    71→  if (p.kind === 'flat') return `خصم ثابت ${(p.flatLyd ?? 0).toFixed(2)} د.ل`
    72→  const cap = p.maxDiscountLyd != null ? ` حتى سقف ${p.maxDiscountLyd.toFixed(2)} د.ل` : ''
    73→  return `خصم ${p.percent}%${cap}`
    74→}

   188→function TopupCodeRow({
   189→  code,
   190→  busy,
   191→  onToggle,
   192→}: {
   193→  code: TopupCodeAdmin
   194→  busy: boolean
   195→  onToggle: (next: boolean) => void
   196→}) {
   197→  const { toast } = useToast()
   198→  const used = !!code.usedByPhone
   199→  const stateLabel = used ? 'مستهلك' : code.isActive ? 'نشط' : 'مُعطّل'
   200→  const stateClass = used
   201→    ? 'bg-[rgba(147,142,126,0.15)] text-[#938E7E]'
   202→    : code.isActive
   203→      ? 'bg-[rgba(16,185,129,0.15)] text-[#10B981]'
   204→      : 'bg-[rgba(245,158,11,0.15)] text-[#F59E0B]'
   205→
   206→  const copy = async () => {
   207→    try {
   208→      await navigator.clipboard.writeText(code.code)
   209→      toast({ title: '📋 نُسخ الكود', description: code.code })
   210→    } catch {
   211→      toast({ title: code.code, description: 'انسخ الكود يدوياً' })
   212→    }
   213→  }
   214→
   215→  return (
   216→    <div className="flex flex-wrap items-center gap-3 rounded-xl border border-[rgba(212,175,55,0.18)] bg-[#041710] px-3 py-2.5">
   217→      <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-[#0A3324] text-lg">
   218→        🎁
   219→      </span>
   220→      <div className="min-w-0 flex-1">
   221→        <p className="font-money ltr-nums text-xs font-black tracking-wider text-[#F5D061]" dir="ltr">
   222→          {code.code}
   223→        </p>
   224→        <p className="mt-0.5 text-[10px] text-[#938E7E]">
   225→          {used ? (
   226→            <>
   227→              استهلكه <span dir="ltr" className="font-money">{code.usedByPhone}</span>
   228→            </>
   229→          ) : (
   230→            code.note || 'كود شحن محفظة — يُستخدم مرة واحدة'
   231→          )}
   232→          {' • '}
   233→          <span className="font-money ltr-nums">{fmtDateTime(code.createdAt)}</span>
   234→        </p>
   235→      </div>
   236→      <Ly amount={code.amountLyd} bold className="text-sm text-[#10B981]" />
   237→      <Badge className={`border-0 px-2 py-0.5 text-[9px] font-bold ${stateClass}`}>{stateLabel}</Badge>
   238→      <div className="flex items-center gap-1.5">
   239→        <button
   240→          onClick={copy}
   241→          aria-label={`نسخ الكود ${code.code}`}
   242→          className="flex h-11 w-11 items-center justify-center rounded-lg text-[#938E7E] transition hover:bg-[rgba(212,175,55,0.12)] hover:text-[#F5D061]"
   243→        >
   244→          <Copy className="h-4 w-4" />
   245→        </button>
   246→        {!used && (
   247→          <Switch
   248→            checked={code.isActive}
   249→            disabled={busy}
   250→            onCheckedChange={onToggle}
   251→            aria-label={`${code.isActive ? 'تعطيل' : 'تفعيل'} كود الشحن ${code.code}`}
   252→          />
   253→        )}
   254→      </div>
   255→    </div>
   256→  )
   257→}
