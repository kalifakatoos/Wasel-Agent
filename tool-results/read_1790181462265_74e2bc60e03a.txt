   146→// ============================================================
   147→// التطبيق الرئيسي
   148→// ============================================================
   149→export default function AdminApp({ onExit }: { onExit: () => void }) {
   150→  const { toast } = useToast()
   151→
   152→  // درع الـ PIN والجلسة
   153→  const [pin, setPin] = useState('')
   154→  const [unlocked, setUnlocked] = useState(false)
   155→  const [checking, setChecking] = useState(false)
   156→  const [errorNonce, setErrorNonce] = useState(0)
   157→
   158→  // بيانات الغرفة
   159→  const [summary, setSummary] = useState<AccountingSummary | null>(null)
   160→  const [vouchersData, setVouchersData] = useState<VouchersPayload | null>(null)
   161→  const [orders, setOrders] = useState<OrderDTO[]>([])
   162→  const [feed, setFeed] = useState<FeedEvent[]>([])
   163→  const [positions, setPositions] = useState<Record<string, FleetPos>>({})
   164→  const [live, setLive] = useState(false)
   165→  const [lastSync, setLastSync] = useState<number | null>(null)
   166→  const [busy, setBusy] = useState<string | null>(null)
   167→  const [tab, setTab] = useState('dashboard')
   168→
   169→  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null)
   170→
   171→  // ---------- سجل الأحداث الحي ----------
   172→  const pushEvent = useCallback((kind: FeedEvent['kind'], text: string, sub?: string) => {
   173→    setFeed((prev) =>
   174→      [
   175→        {
   176→          id: `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   177→          at: Date.now(),
   178→          kind,
   179→          text,
   180→          sub,
   181→        },
   182→        ...prev,
   183→      ].slice(0, 12),
   184→    )
   185→  }, [])
   186→
   187→  // ---------- قفل الجلسة ----------
   188→  const lockSession = useCallback(
   189→    (message?: string) => {
   190→      setUnlocked(false)
   191→      setPin('')
   192→      setSummary(null)
   193→      setVouchersData(null)
   194→      setOrders([])
   195→      setPositions({})
   196→      setFeed([])
   197→      setLive(false)
   198→      setLastSync(null)
   199→      setBusy(null)
   200→      if (message) {
   201→        toast({ title: '🔒 أُقفلت الجلسة الإدارية', description: message, variant: 'destructive' })
   202→      }
   203→    },
   204→    [toast],
   205→  )
   206→
   207→  // ---------- معالج أخطاء موحد (401 = قفل فوري) ----------
   208→  const handleAdminError = useCallback(
   209→    (e: unknown) => {
   210→      if (e instanceof ApiError && e.status === 401) {
   211→        lockSession('رمز الإدارة غير صحيح أو منتهٍ — يلزم إعادة فتح الغرفة')
   212→        return
   213→      }
   214→      toast({
   215→        title: 'تعذر تنفيذ العملية',
   216→        description: e instanceof Error ? e.message : String(e),
   217→        variant: 'destructive',
   218→      })
   219→    },
   220→    [lockSession, toast],
   221→  )
   222→
   223→  // ---------- تحميل بيانات الغرفة (ملخص + سندات + طلبات) ----------
   224→  const loadAll = useCallback(
   225→    async (currentPin: string) => {
   226→      if (!currentPin) return
   227→      const headers = { 'x-admin-pin': currentPin }
   228→      const [sumRes, vouRes, ordRes] = await Promise.allSettled([
   229→        api<AccountingSummary>('/api/accounting/summary', { headers }),
   230→        api<VouchersPayload>('/api/vouchers', { headers }),
   231→        apiGet<{ orders: OrderDTO[] }>('/api/orders?status=all&limit=50'),
   232→      ])
   233→
   234→      for (const r of [sumRes, vouRes]) {
   235→        if (r.status === 'rejected' && r.reason instanceof ApiError && r.reason.status === 401) {
   236→          lockSession('انتهت صلاحية جلسة الإدارة — رمز غير صحيح')
   237→          return
   238→        }
   239→      }
   240→
   241→      if (sumRes.status === 'fulfilled') setSummary(sumRes.value)
   242→      if (vouRes.status === 'fulfilled') setVouchersData(vouRes.value)
   243→      if (ordRes.status === 'fulfilled') setOrders(ordRes.value.orders ?? [])
   244→      setLastSync(Date.now())
   245→    },
   246→    [lockSession],
   247→  )
   248→
   249→  // ---------- فتح الجلسة بالـ PIN ----------
   250→  const attemptUnlock = useCallback(
   251→    async (code: string) => {
   252→      if (code.length !== 4 || checking) return
   253→      setChecking(true)
   254→      try {
   255→        const s = await api<AccountingSummary>('/api/accounting/summary', {
   256→          headers: { 'x-admin-pin': code },
   257→        })
   258→        setPin(code)
   259→        setSummary(s)
   260→        setErrorNonce(0)
   261→        setUnlocked(true)
   262→        setFeed([
   263→          {
   264→            id: `unlock-${Date.now()}`,
   265→            at: Date.now(),
   266→            kind: 'status',
   267→            text: '🔓 فُتحت جلسة غرفة العمليات — البث المالي واللوجستي يعمل الآن',
   268→          },
   269→        ])
   270→        toast({ title: 'أهلاً بك في غرفة العمليات 🛡️', description: 'كل قرش مُساءل — البث الحي متصل' })
   271→      } catch (e) {
   272→        setPin('')
   273→        if (e instanceof ApiError && e.status === 401) {
   274→          setErrorNonce((n) => n + 1)
   275→          toast({ title: 'رمز غير صحيح', description: 'رمز PIN الإداري مرفوض — حاول مجدداً', variant: 'destructive' })
   276→        } else {
   277→          toast({
   278→            title: 'تعذر الاتصال بالخادم',
   279→            description: e instanceof Error ? e.message : 'تحقق من الشبكة وحاول مجدداً',
   280→            variant: 'destructive',
   281→          })
   282→        }
   283→      } finally {
   284→        setChecking(false)
   285→      }
   286→    },
   287→    [checking, toast],
   288→  )
   289→
   290→  // ---------- دورة البيانات: فورية + كل 8 ثوانٍ ----------
   291→  useEffect(() => {
   292→    if (!unlocked || !pin) return
   293→    void loadAll(pin)
   294→    pollRef.current = setInterval(() => {
   295→      void loadAll(pin)
   296→    }, POLL_MS)
   297→    return () => {
   298→      if (pollRef.current) {
   299→        clearInterval(pollRef.current)
   300→        pollRef.current = null
   301→      }
   302→    }
   303→  }, [unlocked, pin, loadAll])
   304→
   305→  // ---------- البث اللحظي: غرفة admin:fleet ----------
   306→  useEffect(() => {
   307→    if (!unlocked) return
   308→    const s = getSocket()
   309→
   310→    const refreshMoney = () => {
   311→      void loadAll(pin)
   312→    }
   313→
   314→    const onConnect = () => {
   315→      setLive(true)
   316→      joinRooms(['admin:fleet'])
   317→      s.emit('fleet:snapshot_request')
   318→    }
   319→    const onDisconnect = () => setLive(false)
   320→
   321→    const onSnapshot = (list: FleetPos[]) => {
   322→      if (!Array.isArray(list)) return
   323→      setPositions((prev) => {
   324→        const next = { ...prev }
   325→        for (const rec of list) if (rec?.driverId) next[rec.driverId] = rec
   326→        return next
   327→      })
   328→    }
   329→
   330→    const onDriverLocation = (rec: FleetPos) => {
   331→      if (!rec?.driverId) return
   332→      setPositions((prev) => ({ ...prev, [rec.driverId]: rec }))
   333→    }
   334→
   335→    const onDriverStatus = (d: { driverId: string; fullName: string; status: DriverDTO['status'] }) => {
   336→      if (!d?.driverId) return
   337→      setPositions((prev) => {
   338→        if (!prev[d.driverId]) return prev
   339→        return {
   340→          ...prev,
   341→          [d.driverId]: { ...prev[d.driverId], status: d.status, updatedAt: new Date().toISOString() },
   342→        }
   343→      })
   344→      setSummary((prev) =>
   345→        prev
   346→          ? { ...prev, drivers: prev.drivers.map((x) => (x.id === d.driverId ? { ...x, status: d.status } : x)) }
   347→          : prev,
   348→      )
   349→      pushEvent('status', `🛵 الكابتن ${d.fullName} أصبح «${STATUS_LABELS[d.status] ?? d.status}»`)
   350→    }
   351→
   352→    const onOrderStatus = (d: { orderId: string; status: string; orderNumber: string }) => {
   353→      if (!d?.orderNumber) return
   354→      pushEvent('status', `📦 الطلب ${d.orderNumber}: ${STATUS_LABELS[d.status] ?? d.status}`)
   355→      setOrders((prev) =>
   356→        prev.map((o) => (o.id === d.orderId ? { ...o, status: d.status as OrderDTO['status'] } : o)),
   357→      )
   358→    }
   359→
   360→    const onRadar = (d: { orderId: string; orderNumber: string; storeName: string; candidates: number }) => {
   361→      if (!d?.orderNumber) return
   362→      pushEvent(
   363→        'radar',
   364→        `📡 رادار: أُرسل عرض الطلب ${d.orderNumber} من «${d.storeName}» إلى ${d.candidates ?? 0} كابتن`,
   365→      )
   366→    }
   367→
   368→    const onCustodySettled = (d: { driverName: string; amount: number; voucherNumber: string }) => {
   369→      if (!d?.voucherNumber) return
   370→      pushEvent(
   371→        'custody',
   372→        `💰 توريد عهدة الكابتن ${d.driverName} بمبلغ ${(d.amount ?? 0).toFixed(2)} د.ل`,
   373→        `سند قبض ${d.voucherNumber}`,
   374→      )
   375→      refreshMoney()
   376→    }
   377→
   378→    const onStorePayout = (d: { storeName: string; amount: number; voucherNumber: string }) => {
   379→      if (!d?.voucherNumber) return
   380→      pushEvent(
   381→        'payout',
   382→        `🏪 صرف مستحقات متجر «${d.storeName}» بمبلغ ${(d.amount ?? 0).toFixed(2)} د.ل`,
   383→        `سند صرف ${d.voucherNumber}`,
   384→      )
   385→      refreshMoney()
   386→    }
   387→
   388→    if (s.connected) onConnect()
   389→    s.on('connect', onConnect)
   390→    s.on('disconnect', onDisconnect)
   391→    s.on('fleet:snapshot', onSnapshot)
   392→    s.on('fleet:driver_location', onDriverLocation)
   393→    s.on('fleet:driver_status', onDriverStatus)
   394→    s.on('order:status_changed', onOrderStatus)
   395→    s.on('order:radar_broadcast', onRadar)
   396→    s.on('driver:custody_settled', onCustodySettled)
   397→    s.on('store:payout_settled', onStorePayout)
   398→
   399→    return () => {
   400→      s.off('connect', onConnect)
   401→      s.off('disconnect', onDisconnect)
   402→      s.off('fleet:snapshot', onSnapshot)
   403→      s.off('fleet:driver_location', onDriverLocation)
   404→      s.off('fleet:driver_status', onDriverStatus)
   405→      s.off('order:status_changed', onOrderStatus)
   406→      s.off('order:radar_broadcast', onRadar)
   407→      s.off('driver:custody_settled', onCustodySettled)
   408→      s.off('store:payout_settled', onStorePayout)
   409→      leaveRooms(['admin:fleet'])
   410→    }
   411→  }, [unlocked, pin, loadAll, pushEvent])
   412→
   413→  // ---------- الإجراءات المالية ----------
   414→  const settleCustody = useCallback(
   415→    async (driver: DriverDTO): Promise<SettleResult | null> => {
   416→      setBusy(`settle:${driver.id}`)
   417→      try {
   418→        const r = await apiPost<SettleResult>(`/api/drivers/${driver.id}/settle`, { pin })
   419→        toast({
   420→          title: `تم توريد عهدة ${r.driver.fullName} للخزينة`,
   421→          description: `سند قبض ${r.voucher.voucherNumber} — ${r.voucher.amountLyd.toFixed(2)} د.ل`,
   422→        })
   423→        void loadAll(pin)
   424→        return r
   425→      } catch (e) {
   426→        handleAdminError(e)
   427→        return null
   428→      } finally {
   429→        setBusy(null)
   430→      }
   431→    },
   432→    [pin, loadAll, handleAdminError, toast],
   433→  )
   434→
   435→  const payoutStore = useCallback(
   436→    async (storeId: string, amount: number): Promise<PayoutStoreResult | null> => {
   437→      setBusy(`payout:${storeId}`)
   438→      try {
   439→        const r = await apiPost<PayoutStoreResult>('/api/accounting/payout-store', {
   440→          pin,
   441→          store_id: storeId,
   442→          amount,
   443→        })
   444→        toast({
   445→          title: 'صُرفت مستحقات المتجر',
   446→          description: `سند صرف ${r.voucher.voucherNumber} — ${r.voucher.amountLyd.toFixed(2)} د.ل • المتبقي ${r.remainingDues.toFixed(2)} د.ل`,
   447→        })
   448→        void loadAll(pin)
   449→        return r
   450→      } catch (e) {
   451→        handleAdminError(e)
   452→        return null
   453→      } finally {
   454→        setBusy(null)
   455→      }
   456→    },
   457→    [pin, loadAll, handleAdminError, toast],
   458→  )
   459→
   460→  const resolveNoShow = useCallback(
   461→    async (order: OrderDTO): Promise<ResolveNoShowResult | null> => {
   462→      setBusy(`noshow:${order.id}`)
   463→      try {
   464→        const r = await apiPost<ResolveNoShowResult>('/api/disputes/resolve-noshow', {
   465→          pin,
   466→          order_id: order.id,
   467→        })
   468→        toast({
   469→          title: `فُضّ نزاع الطلب ${r.orderNumber}`,
   470→          description: `صرف للمتجر ${r.storePayout.toFixed(2)} د.ل • ${r.captainPaid ? `أُودعت أجرة الكابتن ${r.captainFee.toFixed(2)} د.ل لمحفظته` : 'لا يوجد كابتن مُعيَّن — لم تُصرف أي أجرة'} • حُظر الزبون ${r.blockedPhone}`,
   471→        })
   472→        void loadAll(pin)
   473→        return r
   474→      } catch (e) {
   475→        handleAdminError(e)
   476→        return null
   477→      } finally {
   478→        setBusy(null)
   479→      }
   480→    },
   481→    [pin, loadAll, handleAdminError, toast],
   482→  )
   483→
   484→  const closeDay = useCallback(
   485→    async (): Promise<AuditRecord | null> => {
   486→      setBusy('close-day')
   487→      try {
   488→        const r = await apiPost<{ audit: AuditRecord }>('/api/accounting/close-day', { pin })
   489→        toast({
   490→          title: `صدر محضر الجرد ${r.audit.auditCode}`,
   491→          description: 'قُفلت اليومية المالية وأُضيفت لسجل المحاضر',
   492→        })
   493→        void loadAll(pin)
   494→        return r.audit
   495→      } catch (e) {
   496→        handleAdminError(e)
   497→        return null
   498→      } finally {
   499→        setBusy(null)
   500→      }
   501→    },
   502→    [pin, loadAll, handleAdminError, toast],
   503→  )
   504→
