    50→function promoValueLabel(p: PromoAdmin): string {
    51→  if (p.kind === 'flat') return `خصم ثابت ${(p.flatLyd ?? 0).toFixed(2)} د.ل`

   163→function TopupCodeRow({
   164→  code,
   165→  busy,
   166→  onToggle,
   167→}: {
   168→  code: TopupCodeAdmin
   169→  busy: boolean
   170→  onToggle: (next: boolean) => void
   171→}) {
   172→  const { toast } = useToast()
   173→  const used = !!code.usedByPhone
   174→  const stateLabel = used ? 'مستهلك' : code.isActive ? 'نشط' : 'مُعطّل'
   175→  const stateClass = used
   176→    ? 'bg-[rgba(147,142,126,0.15)] text-[#938E7E]'
   177→    : code.isActive
   178→      ? 'bg-[rgba(16,185,129,0.15)] text-[#10B981]'
   179→      : 'bg-[rgba(245,158,11,0.15)] text-[#F59E0B]'
   180→
   181→  const copy = async () => {
   182→    try {
   183→      await navigator.clipboard.writeText(code.code)
   184→      toast({ title: '📋 نُسخ الكود', description: code.code })
   185→    } catch {
   186→      toast({ title: code.code, description: 'انسخ الكود يدوياً' })
   187→    }
   188→  }
   189→
   190→  return (
   191→    <div className="flex flex-wrap items-center gap-3 rounded-xl border border-[rgba(212,175,55,0.18)] bg-[#041710] px-3 py-2.5">
   192→      <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-[#0A3324] text-lg">
   193→        🎁
   194→      </span>
   195→      <div className="min-w-0 flex-1">
   196→        <p className="font-money ltr-nums text-xs font-black tracking-wider text-[#F5D061]" dir="ltr">
   197→          {code.code}
   198→        </p>
   199→        <p className="mt-0.5 text-[10px] text-[#938E7E]">
   200→          {used ? (
   201→            <>
   202→              استهلكه <span dir="ltr" className="font-money">{code.usedByPhone}</span>
   203→            </>
   204→          ) : (
   205→            code.note || 'كود شحن محفظة — يُستخدم مرة واحدة'
   206→          )}
   207→          {' • '}
   208→          <span className="font-money ltr-nums">{fmtDateTime(code.createdAt)}</span>
   209→        </p>
   210→      </div>
   211→      <Ly amount={code.amountLyd} bold className="text-sm text-[#10B981]" />
   212→      <Badge className={`border-0 px-2 py-0.5 text-[9px] font-bold ${stateClass}`}>{stateLabel}</Badge>
   213→      <div className="flex items-center gap-1.5">
   214→        <button
   215→          onClick={copy}
   216→          aria-label={`نسخ الكود ${code.code}`}
   217→          className="flex h-11 w-11 items-center justify-center rounded-lg text-[#938E7E] transition hover:bg-[rgba(212,175,55,0.12)] hover:text-[#F5D061]"
   218→        >
   219→          <Copy className="h-4 w-4" />
   220→        </button>
   221→        {!used && (
   222→          <Switch
   223→            checked={code.isActive}
   224→            disabled={busy}
   225→            onCheckedChange={onToggle}
   226→            aria-label={`${code.isActive ? 'تعطيل' : 'تفعيل'} كود الشحن ${code.code}`}
   227→          />
   228→        )}
   229→      </div>
   230→    </div>
   231→  )
   232→}
