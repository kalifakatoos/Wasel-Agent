   146→function TopupCodeRow({
   147→  code,
   148→  busy,
   149→  onToggle,
   150→}: {
   151→  code: TopupCodeAdmin
   152→  busy: boolean
   153→  onToggle: (next: boolean) => void
   154→}) {
   155→  const { toast } = useToast()
   156→  const used = !!code.usedByPhone
   157→  const stateLabel = used ? 'مستهلك' : code.isActive ? 'نشط' : 'مُعطّل'
   158→  const stateClass = used
   159→    ? 'bg-[rgba(147,142,126,0.15)] text-[#938E7E]'
   160→    : code.isActive
   161→      ? 'bg-[rgba(16,185,129,0.15)] text-[#10B981]'
   162→      : 'bg-[rgba(245,158,11,0.15)] text-[#F59E0B]'
   163→
   164→  const copy = async () => {
   165→    try {
   166→      await navigator.clipboard.writeText(code.code)
   167→      toast({ title: '📋 نُسخ الكود', description: code.code })
   168→    } catch {
   169→      toast({ title: code.code, description: 'انسخ الكود يدوياً' })
   170→    }
   171→  }
   172→
   173→  return (
   174→    <div className="flex flex-wrap items-center gap-3 rounded-xl border border-[rgba(212,175,55,0.18)] bg-[#041710] px-3 py-2.5">
   175→      <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-[#0A3324] text-lg">
   176→        🎁
   177→      </span>
   178→      <div className="min-w-0 flex-1">
   179→        <p className="font-money ltr-nums text-xs font-black tracking-wider text-[#F5D061]" dir="ltr">
   180→          {code.code}
   181→        </p>
   182→        <p className="mt-0.5 text-[10px] text-[#938E7E]">
   183→          {used ? (
   184→            <>
   185→              استهلكه <span dir="ltr" className="font-money">{code.usedByPhone}</span>
   186→            </>
   187→          ) : (
   188→            code.note || 'كود شحن محفظة — يُستخدم مرة واحدة'
   189→          )}
   190→          {' • '}
   191→          <span className="font-money ltr-nums">{fmtDateTime(code.createdAt)}</span>
   192→        </p>
   193→      </div>
   194→      <Ly amount={code.amountLyd} bold className="text-sm text-[#10B981]" />
   195→      <Badge className={`border-0 px-2 py-0.5 text-[9px] font-bold ${stateClass}`}>{stateLabel}</Badge>
   196→      <div className="flex items-center gap-1.5">
   197→        <button
   198→          onClick={copy}
   199→          aria-label={`نسخ الكود ${code.code}`}
   200→          className="flex h-11 w-11 items-center justify-center rounded-lg text-[#938E7E] transition hover:bg-[rgba(212,175,55,0.12)] hover:text-[#F5D061]"
   201→        >
   202→          <Copy className="h-4 w-4" />
   203→        </button>
   204→        {!used && (
   205→          <Switch
   206→            checked={code.isActive}
   207→            disabled={busy}
   208→            onCheckedChange={onToggle}
   209→            aria-label={`${code.isActive ? 'تعطيل' : 'تفعيل'} كود الشحن ${code.code}`}
   210→          />
   211→        )}
   212→      </div>
   213→    </div>
   214→  )
   215→}
